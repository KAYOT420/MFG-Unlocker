cmake_minimum_required(VERSION 3.24)
project(IntelNativeTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
enable_testing()
if(NOT EXISTS "${XESS_SDK_ROOT}/inc/xess_fg/xefg_swapchain_d3d12.h")
  message(FATAL_ERROR "XESS_SDK_ROOT must identify the pinned SDK")
endif()
add_executable(IntelNativeTests main.cpp ../../source/native/xess/intel_native.cpp)
target_include_directories(IntelNativeTests PRIVATE "${XESS_SDK_ROOT}/inc/xess_fg")
target_compile_definitions(IntelNativeTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(IntelNativeTests PRIVATE /W4 /EHsc)
add_test(NAME native_driver_contract COMMAND IntelNativeTests "${INTEL_FG_FIXTURE}" "${INTEL_LL_FIXTURE}" "${XESS_FG_FIXTURE}")
add_library(WrongNativeXeLL SHARED wrong_xell.cpp)
set_target_properties(WrongNativeXeLL PROPERTIES OUTPUT_NAME wrong-native-xell)
add_executable(IntelNativeSelectionTests selection.cpp ../../source/native/xess/intel_native.cpp)
target_include_directories(IntelNativeSelectionTests PRIVATE "${XESS_SDK_ROOT}/inc/xess_fg")
target_compile_definitions(IntelNativeSelectionTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(IntelNativeSelectionTests PRIVATE /W4 /EHsc)
add_dependencies(IntelNativeSelectionTests WrongNativeXeLL)
# arcnative.3: XeLL pairing follows the driver's binding rule and loads nothing.
foreach(mode builtin handoff ungated missing not-sibling wrong-libxell)
  add_test(NAME native_xell_${mode} COMMAND IntelNativeSelectionTests "${mode}"
    "${INTEL_FG_FIXTURE}" "${XESS_LL_FIXTURE}" "${XESS_LL_UNGATED_FIXTURE}" "$<TARGET_FILE:WrongNativeXeLL>"
    "${CMAKE_CURRENT_BINARY_DIR}/selection-fixtures/${mode}")
endforeach()
